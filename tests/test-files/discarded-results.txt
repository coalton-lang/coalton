1 A discarded Result is a warning
================================================================================

(package coalton-unit-tests/discarded-results
  (import coalton-prelude))

(declare check (Boolean -> (Result String Unit)))
(define (check ok?)
  (if ok? (Ok Unit) (Err "failed")))

(define (twice)
  (check True)
  (check False))

2 The last form of a when body is discarded

================================================================================
3 A Result can be discarded explicitly
================================================================================

(package coalton-unit-tests/discarded-results
  (import coalton-prelude))

(declare check (Boolean -> (Result String Unit)))
(define (check ok?)
  (if ok? (Ok Unit) (Err "failed")))

(define (twice)
  (let _ = (check True))
  (check False))

================================================================================
5 A Result in a protect cleanup can be discarded explicitly
================================================================================

(package coalton-unit-tests/discarded-results
  (import coalton-prelude))

(declare check (Boolean -> (Result String Unit)))
(define (check ok?)
  (if ok? (Ok Unit) (Err "failed")))

(declare checked (Integer -> Integer))
(define (checked x)
  (protect x
    (let _ = (check False))
    Unit))
