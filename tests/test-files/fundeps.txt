118 Subclasses can agree on a hidden superclass dependent
================================================================================

(package coalton-unit-test/fundep-siblings)

(define-class (Parent :x :y (:x -> :y))
  (get-y (:x -> :y)))
(define-class (Parent :x :y => LeftChild :x)
  (left-id (:x -> :x)))
(define-class (Parent :x :y => RightChild :x)
  (right-id (:x -> :x)))

(declare both ((LeftChild :x) (RightChild :x) => :x -> :x))
(define (both x) (left-id (right-id x)))

(declare expose ((Parent :x :result) (LeftChild :x) => :x -> :result))
(define (expose x) (get-y (left-id x)))
