================================================================================
100 Native exceptions
================================================================================

(package test-package)

(repr :native cl:error)
(define-exception LispError
  "Any Lisp error.")

(repr :native cl:division-by-zero)
(define-exception DivisionByZero)

(declare safe-divide (Integer * Integer -> Integer))
(define (safe-divide a b)
  (catch (lisp (-> Integer) (a b) (cl:/ a b))
    ((the DivisionByZero _) 0)
    ((the LispError e) (throw e))))

(declare rethrow (LispError -> Integer))
(define (rethrow e)
  (throw e))
